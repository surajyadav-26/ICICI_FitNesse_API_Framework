    localStorage.setItem(USER_STORE_KEY, JSON.stringify(users));

      method: "POST",

      body: JSON.stringify(users)

      console.warn("User file store unavailable; browser cache retained:", error);
    });

      if (users && typeof users === "object") {
        localStorage.setItem(USER_STORE_KEY, JSON.stringify(users));
      }
    })
    .catch(function(error) {
      console.warn("User file store unavailable; using browser cache:", error);

  function renderUserTable() {

      var user = tempUsers[username];

      nameCell.textContent = username;
      row.appendChild(nameCell);

      var displayCell = document.createElement("td");
      var displayInput = document.createElement("input");
      displayInput.className = "ici-env-input";
      displayInput.value = user.display || username;

  window.iciSaveUserManagement = function() {
    if (!currentSessionIsAdmin()) {
      alert("Only Admin users can save user changes.");
      return;
    }

    var errorEl = document.getElementById("ici-user-management-error");
    var adminCount = 0;
    Object.keys(tempUsers).forEach(function(username) {
      var displayInput = document.querySelector("[data-user-display='" + username + "']");
      var roleSelect = document.querySelector("[data-user-role='" + username + "']");
      if (displayInput) tempUsers[username].display = displayInput.value.trim() || username;
      if (roleSelect) tempUsers[username].role = roleSelect.value;
      if (tempUsers[username].role === "admin") adminCount++;
    });

    var username = (document.getElementById("ici-add-user-name").value || "").trim().toLowerCase();
    var display = (document.getElementById("ici-add-user-display").value || "").trim();
    var password = document.getElementById("ici-add-user-password").value || "";
    var role = document.getElementById("ici-add-user-role").value;

    if (username || display || password) {
      if (!/^[a-z0-9._-]+$/.test(username) || !password || !display) {
        errorEl.textContent = "Enter a valid username, display name, and password.";
        return;
      }
      if (tempUsers[username]) {
        errorEl.textContent = "That username already exists.";
        return;
      }
      tempUsers[username] = { password: password, role: role, display: display };
      if (role === "admin") adminCount++;
    }

    if (adminCount < 1) {
      errorEl.textContent = "At least one Admin user must remain.";
      return;
    }
    saveUsers(tempUsers);
    iciCloseUserManagement();
    var session = loadSession();
    if (session && tempUsers[session.user]) {
      saveSession(session.user, tempUsers[session.user].role, tempUsers[session.user].display);
      showApp(tempUsers[session.user]);
    }
  };

  /* ── Zero-Dependency Dynamic Table-to-Excel/CSV Exporter! ── */
  window.iciExportTableToCSV = function() {
    var tables = document.querySelectorAll("table.fit_table, table.statistics, .history table, .test-history table, table");
    if (tables.length === 0) {
      alert("No tabular data found on this page to export.");
      return;
    }

    var csvRows = [];
    tables.forEach(function(table, tableIdx) {
      if (tableIdx > 0) csvRows.push(""); // separation spacer
      
      var rows = table.querySelectorAll("tr");
      rows.forEach(function(row) {
        var cols = row.querySelectorAll("th, td");
        var rowData = [];
        
        cols.forEach(function(col) {
          var text = col.textContent.replace(/\s+/g, " ").replace(/"/g, '""').trim();
          rowData.push('"' + text + '"');
        });
        
        csvRows.push(rowData.join(","));
      });
    });

    var csvString = csvRows.join("\n");
    var blob = new Blob([csvString], { type: "text/csv;charset=utf-8;" });
    var link = document.createElement("a");
    
    var pageTitle = (document.title || "Report").replace(/[^a-zA-Z0-9]/g, "_");
    link.download = pageTitle + "_Report.csv";
    link.href = URL.createObjectURL(blob);
    link.style.display = "none";
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
  };

      a.className = "icici-nav-btn" + (l.adminOnly ? " admin-only" : "");
      a.textContent = l.label;
      nav.appendChild(a);
    });
  }

  function showApp(session) {
    try {
      if (!session || !session.role) {
        throw new Error("Invalid session data");
      }

      var overlay = document.getElementById("icici-login-overlay");
      if (overlay) overlay.style.display = "none";
      document.body.style.overflow = ""; // Restore body scrolling!

      var banner = document.getElementById("icici-header-banner");
      if (banner) banner.style.display = "flex";

      var nameEl = document.getElementById("icici-user-display");
      if (nameEl) nameEl.textContent = session.display || "User";

      var badge = document.getElementById("icici-role-badge");
      if (badge) {
        badge.textContent = session.role.charAt(0).toUpperCase() + session.role.slice(1);
        badge.className = "icici-role-badge role-" + session.role;
      }

      var userManagementMenu = document.getElementById("ici-user-management-menu");
      if (userManagementMenu) {
        userManagementMenu.classList.toggle("ici-user-management-hidden", session.role !== "admin");
      }

      buildNav(session.role);
      populateEnvUI(); // Load Environment selector cleanly on log in!

      if (session.role === "viewer") {
        lockForReadOnly();
      }
    } catch(e) {
      console.warn("Clearing corrupted session:", e);
      clearSession();
      var overlay = document.getElementById("icici-login-overlay");
      if (overlay) overlay.style.display = "flex";
      document.body.style.overflow = "hidden";
    }
  }

  function lockForReadOnly() {
    function applyViewerLock() {
      ["input[type=submit]","button[type=submit]",
       "textarea","input[type=text]"].forEach(function(sel) {
        document.querySelectorAll(sel).forEach(function(el) {
          el.disabled = true;
          el.title = "Read-only access — contact Admin to edit";
        });
      });
      document.querySelectorAll("a[href*='?new'], a[href*='?edit'], a[href*='?delete'], a[href*='?properties'], a[href*='?refactor'], #ici-nav-env-container").forEach(function(el) {
        el.style.display = "none";
      });
      if (/[?&](edit|addChild|saveData|save|delete|properties)/.test(location.search)) {
        document.body.innerHTML =
          '<div style="margin:80px auto;max-width:500px;text-align:center;font-family:Arial">' +
          '<h2 style="color:#A6192E">Access Denied</h2>' +
          '<p>Your role (<strong>Viewer</strong>) does not have permission to perform this action.</p>' +
          '<a href="/" style="color:#004A80">← Back to Home</a></div>';
      }
    }
    if (document.readyState === "loading") {
      document.addEventListener("DOMContentLoaded", applyViewerLock);
    } else {
      applyViewerLock();
    }
  }

  window.iciLogin = function() {
    var u = (document.getElementById("icici-username").value || "").trim().toLowerCase();
    var p = document.getElementById("icici-password").value || "";
    var errEl = document.getElementById("icici-login-error");

    if (!u || !p) { errEl.textContent = "Please enter username and password."; return; }

      return;
    }

    errEl.textContent = "";
    saveSession(u, rec.role, rec.display);
    showApp({ display: rec.display, role: rec.role });
  };

  function loadConfig() {
    var stored = localStorage.getItem(STORAGE_KEY);
    if (!stored) {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(DEFAULT_CONFIG));
      return DEFAULT_CONFIG;
    }
    try { 
      var parsed = JSON.parse(stored);
      // Auto-migrate old string-only configurations seamlessly to support both API and UI URLs!
      var migrated = false;
      for (var env in parsed.environments) {
        if (typeof parsed.environments[env] === "string") {
          var oldUrl = parsed.environments[env];
          var defaultUi = DEFAULT_CONFIG.environments[env] ? DEFAULT_CONFIG.environments[env].ui : "https://retailnetbanking.icici.bank.in/login-page";
          parsed.environments[env] = { api: oldUrl, ui: defaultUi };
          migrated = true;
        }
      }
      if (migrated) {
        localStorage.setItem(STORAGE_KEY, JSON.stringify(parsed));
      }
      return parsed; 
    } catch(e) { return DEFAULT_CONFIG; }
  }

  function saveConfig(config) {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(config));

  window.iciSwitchEnv = function(selectedEnv) {
    var config = loadConfig();
    if (!config.environments[selectedEnv]) return;
    
    config.active = selectedEnv;
    saveConfig(config);
    
    // Write both API_BASE_URL and UI_BASE_URL dynamically to FitNesse VariablePage!
    var activeUrl = config.environments[selectedEnv].api || "";
    var activeUiUrl = config.environments[selectedEnv].ui || "https://retailnetbanking.icici.bank.in/login-page";
    
    // Force strictly sequential suite-by-suite executions to prevent parallel thread clashing!
    var parallelStr = "!define runInParallel {false}\n!define fitnesse.runner.parallel {1}\n";
    
    var formData = new URLSearchParams();
    formData.append("responder", "saveData");
    formData.append("pageContent", "!define API_BASE_URL {" + activeUrl + "}\n!define UI_BASE_URL {" + activeUiUrl + "}\n" + parallelStr);
    
    fetch("/VariablePage", {
      method: "POST",
      body: formData,
      headers: { "Content-Type": "application/x-www-form-urlencoded" }
    })
    .then(function() {
      location.reload();
    })
    .catch(function(err) {
      console.error("Failed to write to VariablePage:", err);
      location.reload();
    });
  };

  function renderModalTable() {
    var tbody = document.getElementById("ici-modal-table-body");
    if (!tbody) return;
    tbody.innerHTML = "";
    
    for (var env in tempEnvironments) {
      var tr = document.createElement("tr");
      
      var tdName = document.createElement("td");
      tdName.style.fontWeight = "bold";
      tdName.textContent = env;
      tr.appendChild(tdName);
      
      // API URL Input column
      var tdUrl = document.createElement("td");
      var inputApi = document.createElement("input");
      inputApi.type = "text";
      inputApi.className = "ici-env-input";
      inputApi.value = tempEnvironments[env].api || "";
      inputApi.setAttribute("data-env-api", env);
      inputApi.onchange = function(e) {
        var envName = e.target.getAttribute("data-env-api");
        if (!tempEnvironments[envName]) tempEnvironments[envName] = {};
        tempEnvironments[envName].api = e.target.value.trim();
      };
      tdUrl.appendChild(inputApi);
      tr.appendChild(tdUrl);
      
      // UI URL Input column
      var tdUiUrl = document.createElement("td");
      var inputUi = document.createElement("input");
      inputUi.type = "text";
      inputUi.className = "ici-env-input";
      inputUi.value = tempEnvironments[env].ui || "";
      inputUi.setAttribute("data-env-ui", env);

  /* ── Bulletproof Favicon Override with MutationObserver ── */
  function overrideFavicon() {
    var faviconUrl = "/favicon.ico";
    
    // Resolve faviconUrl to its absolute form once to prevent relative-to-absolute infinite loop triggers!
    var tempA = document.createElement("a");
    tempA.href = faviconUrl;
    var absoluteFaviconUrl = tempA.href;
    var normTarget = (absoluteFaviconUrl || "").split("?")[0].replace(/^https?:/, "").toLowerCase();
    
    function lockLink(link) {
      if (!link || link._locked) return;
      link._locked = true;
      
      // 1. Force-lock ID
      link.id = "favicon";
      
      // 2. Intercept and completely block setAttribute("href", ...) and setAttribute("type", ...)
      try {
        var originalSetAttribute = link.setAttribute;
        link.setAttribute = function(name, val) {
          if (name === "href" || name === "type" || name === "rel") {
            return; // Block any changes to these crucial attributes!
          }
          originalSetAttribute.call(this, name, val);
        };
      } catch (e) {}
      
      // 3. Lock href and type properties at the JS engine compiler level using getters and setters!
      try {
        Object.defineProperty(link, 'href', {
          get: function() { return absoluteFaviconUrl; },
          set: function(val) { /* Blocked! */ },
          configurable: true
        });
        
        Object.defineProperty(link, 'type', {
          get: function() { return "image/x-icon"; },
          set: function(val) { /* Blocked! */ },
          configurable: true
        });
      } catch (e) {}
    }
    
    function apply() {
      var links = document.querySelectorAll("link[rel*='icon']");
      
      if (links.length === 0) {
        var iconLink = document.createElement("link");
        iconLink.rel = "icon";
        iconLink.href = absoluteFaviconUrl;
        lockLink(iconLink);
        if (document.head) {
          document.head.appendChild(iconLink);
        } else {
          document.getElementsByTagName("head")[0].appendChild(iconLink);
        }
        
        var shortcutLink = document.createElement("link");
        shortcutLink.rel = "shortcut icon";
        shortcutLink.href = absoluteFaviconUrl;
        lockLink(shortcutLink);
        if (document.head) {
          document.head.appendChild(shortcutLink);
        } else {
          document.getElementsByTagName("head")[0].appendChild(shortcutLink);
        }
      } else {
        links.forEach(function(link) {
          lockLink(link);
          
          // One-time inline sync in case it was already altered before we locked it
          var normCurrent = (link.href || "").split("?")[0].replace(/^https?:/, "").toLowerCase();
          var normTarget = (absoluteFaviconUrl || "").split("?")[0].replace(/^https?:/, "").toLowerCase();
          if (normCurrent !== normTarget) {
            try {
              link.href = absoluteFaviconUrl;
            } catch(e) {}
          }
        });
      }
    }

    apply();
    // Run on key loading stages
    document.addEventListener("DOMContentLoaded", apply);
    window.addEventListener("load", apply);
    
    // Set up MutationObserver to block child and attribute attempts to overwrite our favicon!
    var target = document.documentElement;
    if (target && typeof MutationObserver !== 'undefined') {
      var observer = new MutationObserver(function(mutations) {
        var needsApply = false;
        mutations.forEach(function(mutation) {
          if (mutation.type === 'childList') {
            var addedNodes = Array.prototype.slice.call(mutation.addedNodes);
            var hasIcon = addedNodes.some(function(node) {
              return node.nodeType === 1 && node.tagName.toUpperCase() === 'LINK' && node.getAttribute('rel') && node.getAttribute('rel').indexOf('icon') !== -1;
            });
            if (hasIcon) needsApply = true;
          } else if (mutation.type === 'attributes') {
            var node = mutation.target;
            if (node && node.nodeType === 1 && node.tagName.toUpperCase() === 'LINK') {
              var rel = node.getAttribute('rel');
              if (rel && rel.indexOf('icon') !== -1) {
                var normCurrent = (node.href || "").split("?")[0].replace(/^https?:/, "").toLowerCase();
                if (normCurrent !== normTarget) {
                  needsApply = true;
                }
              }
            }
          }
        });
        if (needsApply) {
          apply();
        }
      });
      observer.observe(target, { 
        childList: true, 
        subtree: true, 
        attributes: true
      });
    }
    
    // 200ms high-frequency heartbeat sweep to guarantee absolute victory over FitNesse scripts
    setInterval(apply, 200);
  }

          if (value !== defaultValue && value.trim() !== '') {
            hasData = true;
            break;
          }
        }
        
        formDataChanged[formId] = hasData;
      });
    });
    
    // Fix all cancel links (FitNesse uses <a class="button">Cancel</a>)
    var cancelLinks = document.querySelectorAll('a.button, a[href]:not([href="#"])');
    
    cancelLinks.forEach(function(link) {

        console.log('ICICI UX: Found cancel link:', link.href);
        
        link.addEventListener('click', function(e) {
          e.preventDefault();
          
          // Find the closest form to check if it has changes
          var closestForm = link.closest('form') || document.querySelector('form');
          var formIndex = Array.prototype.indexOf.call(forms, closestForm);
          var formId = 'form_' + formIndex;
          var hasChanges = formDataChanged[formId] || false;
          
          if (!hasChanges) {
            // No changes, just go back
            console.log('ICICI UX: No changes detected, navigating back');
            window.history.back();
          } else {
            // Ask for confirmation
            if (confirm('You have unsaved changes. Discard them?')) {
              console.log('ICICI UX: User confirmed discard, navigating back');
              window.history.back();
            }
          }
          
          return false;
        });
      }
    });
    
    /* ────────────────────────────────────────────────────────────
       FIX 2: SMART RADIO BUTTON DEFAULTS
       ──────────────────────────────────────────────────────────── */
    // FitNesse uses name="PageType" with capitals!
    var pageTypeRadios = document.getElementsByName('PageType');

      var currentPath = window.location.pathname;

      console.log('  Path:', currentPath);

      console.log('  Radio count:', pageTypeRadios.length);
      
      // Check current state
      for (var j = 0; j < pageTypeRadios.length; j++) {

      }
      
      // Detect context
      var isAddingChild = currentSearch.indexOf('responder=addChild') !== -1 || currentSearch.indexOf('addChild') !== -1;
      var isProperties = currentSearch.indexOf('properties') !== -1;
      var isInSuite = currentPath.match(/\/(Sanity|Smoke|DummyAPI|FrontPage)/i);

        console.log('  → Properties page detected, keeping current selection');
        return;

      if (isAddingChild && isInSuite) {
        // Adding child to a suite → default to Test
        defaultType = 'Test';
        console.log('  → Context: Adding child to suite → Defaulting to TEST');
      } else {
        console.log('  → Context: New page → Defaulting to SUITE');
      }
      
      // Apply selection
      for (var i = 0; i < pageTypeRadios.length; i++) {
        if (pageTypeRadios[i].value === defaultType) {
          pageTypeRadios[i].checked = true;
          console.log('%c  ✓ Auto-selected:', 'color: green; font-weight: bold;', defaultType);
          
          // Trigger change event
          if (typeof Event === 'function') {
            pageTypeRadios[i].dispatchEvent(new Event('change', { bubbles: true }));
          }
          break;
        }
      }
    } else {
      console.log('No PageType radio buttons found on this page');
    }
    
    /* ────────────────────────────────────────────────────────────
       FIX 3: UNIVERSAL ICICI THEME
       ──────────────────────────────────────────────────────────── */
    if (document.body && !document.body.classList.contains('icici-theme')) {
      document.body.classList.add('icici-theme');
      console.log('ICICI UX: Applied icici-theme class to body');
    }
    
    // Watch for dynamically added content
    if (typeof MutationObserver !== 'undefined') {
      var themeObserver = new MutationObserver(function(mutations) {
        mutations.forEach(function(mutation) {
          if (mutation.addedNodes) {
            for (var i = 0; i < mutation.addedNodes.length; i++) {
              var node = mutation.addedNodes[i];
              if (node.nodeType === 1) {
                // Apply theme classes to new elements
                if (node.tagName === 'BUTTON' || (node.tagName === 'INPUT' && node.type === 'submit')) {
                  node.classList.add('icici-styled');
                }
                if (node.tagName === 'TABLE') {
                  node.classList.add('icici-table');
                }
              }
            }
          }
        });
      });
      
      themeObserver.observe(document.body, { childList: true, subtree: true });
      console.log('ICICI UX: Theme observer active');
    }

       FIX 6: PREVENT DUPLICATE RESPONDER FORM SERIALIZATION
       ──────────────────────────────────────────────────────────── */
    document.addEventListener("submit", function(e) {
      var form = e.target;
      if (form && form.tagName === "FORM") {
        var action = form.getAttribute("action") || "";
        // If action already has a responder query parameter, prevent duplicate form fields
        if (action.indexOf("responder=") !== -1) {
          var hiddenResponders = form.querySelectorAll("input[type='hidden'][name='responder']");
          hiddenResponders.forEach(function(input) {
            input.removeAttribute("name"); // Prevents the browser from serializing this element!
            console.log("ICICI UX: Suppressed duplicate hidden responder input name to avoid Java errors.");
          });
        }
      }
    });

          '<button class="ici-dropdown-item" onclick="iciOpenAllureReport()"><span class="ici-dropdown-icon">📈</span><span>Allure Report</span></button>';
        document.body.appendChild(dropdown);
      }

      if (dropdown.style.display === "none" || !dropdown.style.display) {

    window.iciOpenAllureReport = function() {
      // Prevent opening Allure Report prematurely if a test suite is actively running!
      var stopBtn = document.getElementById("stop-test") || document.querySelector(".stop-test");
      if (!stopBtn) {
        var els = document.querySelectorAll("input, button, a");
        for (var idx = 0; idx < els.length; idx++) {
          var val = (els[idx].value || els[idx].textContent || "").trim();
          if (val.indexOf("Stop") > -1) {
            stopBtn = els[idx];
            break;
          }
        }
      }
      if (stopBtn) {
        alert("⚠️ Suite execution is still in progress!\n\nPlease wait for the FitNesse run to completely finish (turn green/red) before opening your Allure Report.");
        return;
      }

      var repDropdown = document.getElementById("ici-report-dropdown");
      if (repDropdown) repDropdown.style.display = "none";

      var progressSpinner = document.getElementById("ici-header-progress");
      if (progressSpinner) progressSpinner.style.display = "flex";

      fetch("http://127.0.0.1:8090/serve-allure")
      .then(function(r) { return r.json(); })
      .then(function(data) {
        if (progressSpinner) progressSpinner.style.display = "none";
        if (data.error) {
          alert("Allure Launch Failed:\n\nAllure command-line utility was not found in your system PATH!\n\nTo install Allure globally on Windows, open your command prompt and run:\nnpm install -g allure-commandline\n\nOnce installed, restart start_server.bat and click Allure Report again!");
        } else if (data.url) {
          // Open the permanent, local FitNesse-hosted Allure dashboard directly on port 8080!
          window.open(data.url, "_blank");
        }
      })
      .catch(function(err) {
        if (progressSpinner) progressSpinner.style.display = "none";
        console.error("Allure launch request failed:", err);
        alert("Failed to connect to local server on port 8090. Make sure your start_server.bat is active!");
      });
    };

    window.iciCopyToClipboard = function(text) {
      if (navigator.clipboard) {
        navigator.clipboard.writeText(text).then(function() {
          alert("📋 Copied to clipboard: " + text);
        });
      } else {
        var textarea = document.createElement("textarea");
        textarea.value = text;
        document.body.appendChild(textarea);
        textarea.select();
        document.execCommand("copy");
        document.body.removeChild(textarea);
        alert("📋 Copied to clipboard: " + text);
      }
    };

    window.iciOnBuilderActionChange = function() {
      var action = document.getElementById("ici-builder-action").value;
      var locatorContainer = document.getElementById("ici-builder-locator-container");
      var param1Container = document.getElementById("ici-builder-param-1-container");
      var param2Container = document.getElementById("ici-builder-param-2-container");
      var param1Label = document.getElementById("ici-builder-param-1-label");
      var param2Label = document.getElementById("ici-builder-param-2-label");
      var param1Input = document.getElementById("ici-builder-param-1");
      var param2Input = document.getElementById("ici-builder-param-2");

      if (action === "navigate") {
        locatorContainer.style.display = "none";
        param1Container.style.display = "block";
        param2Container.style.display = "none";
        param1Label.textContent = "Target URL";
        param1Input.placeholder = "e.g. https://www.saucedemo.com/";
      } else if (action === "fill") {
        locatorContainer.style.display = "block";
        param1Container.style.display = "none"; // Handled by locator container
        param2Container.style.display = "block";
        param2Label.textContent = "Value to Type";
        param2Input.placeholder = "e.g. standard_user";
        iciOnBuilderLocatorChange(); // Update locator parameters
      } else if (action === "click") {
        locatorContainer.style.display = "block";
        param1Container.style.display = "none";
        param2Container.style.display = "none";
        iciOnBuilderLocatorChange();
      } else if (action === "wait") {
        locatorContainer.style.display = "none";
        param1Container.style.display = "block";
        param2Container.style.display = "block";
        param1Label.textContent = "Text Content to wait for";
        param1Input.placeholder = "e.g. Products";
        param2Label.textContent = "Timeout (ms)";
        param2Input.placeholder = "e.g. 5000";
      } else if (action === "verify") {
        locatorContainer.style.display = "none";
        param1Container.style.display = "block";
        param2Container.style.display = "none";
        param1Label.textContent = "Expected Text";
        param1Input.placeholder = "e.g. Products";
      }
      iciGenerateBuilderOutput();
    };

    window.iciGenerateBuilderOutput = function() {
      var action = document.getElementById("ici-builder-action").value;
      var param1 = document.getElementById("ici-builder-param-1").value || "";
      var param2 = document.getElementById("ici-builder-param-2").value || "";
      
      // Determine the locator string (either selected from dropdown or custom typed)
      var locatorStr = "";
      if (action === "fill" || action === "click") {
        var selectVal = document.getElementById("ici-builder-locator-select").value;
        if (selectVal === "custom") {
          locatorStr = document.getElementById("ici-builder-locator-custom").value || "";
        } else {
          locatorStr = selectVal;
        }
      }

      var output = "";
      if (action === "navigate") {
        output = "| navigate to | " + param1 + " |";
      } else if (action === "fill") {
        output = "| fill field | " + locatorStr + " | with value | " + param2 + " |";
      } else if (action === "click") {
        output = "| click button | " + locatorStr + " |";
      } else if (action === "wait") {
        output = "| wait for text | " + param1 + " | timeout | " + (param2 || "5000") + " |";
      } else if (action === "verify") {
        output = "| check | verify text present | " + param1 + " | true |";
      } else if (action === "close") {
        output = "| close browser |";
      }
      
      // Update single step preview
      document.getElementById("ici-builder-step-preview").textContent = output;
    };

      }
      
      // Append the new step as a clean line
      currentVal += "\n" + step;
      scriptBox.value = currentVal;
    };
