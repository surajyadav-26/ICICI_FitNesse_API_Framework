  /* ── Bulletproof Favicon Override with MutationObserver ── */
  function overrideFavicon() {
    var faviconUrl = "/files/fitnesse/icici/img/icici-favicon.ico";
    
    // Resolve faviconUrl to its absolute form once to prevent relative-to-absolute infinite loop triggers!
    var tempA = document.createElement("a");
    tempA.href = faviconUrl;
    var absoluteFaviconUrl = tempA.href;
    var normTarget = (absoluteFaviconUrl || "").split("?")[0].replace(/^https?:/, "").toLowerCase();
    
    function lockLink(link) {
      if (!link || link._locked) return;
      link._locked = true;
      
      // 1. Force-lock ID
      link.id = "favicon";
      
      // 2. Intercept and completely block setAttribute("href", ...) and setAttribute("type", ...)
      try {
        var originalSetAttribute = link.setAttribute;
        link.setAttribute = function(name, val) {
          if (name === "href" || name === "type" || name === "rel") {
            return; // Block any changes to these crucial attributes!
          }
          originalSetAttribute.call(this, name, val);
        };
      } catch (e) {}
      
      // 3. Lock href and type properties at the JS engine compiler level using getters and setters!
      try {
        Object.defineProperty(link, 'href', {
          get: function() { return absoluteFaviconUrl; },
          set: function(val) { /* Blocked! */ },
          configurable: true
        });
        
        Object.defineProperty(link, 'type', {
          get: function() { return "image/x-icon"; },
          set: function(val) { /* Blocked! */ },
          configurable: true
        });
      } catch (e) {}
    }
    
    function apply() {
      var links = document.querySelectorAll("link[rel*='icon']");
      
      if (links.length === 0) {
        var iconLink = document.createElement("link");
        iconLink.rel = "icon";
        iconLink.href = absoluteFaviconUrl;
        lockLink(iconLink);
        if (document.head) {
          document.head.appendChild(iconLink);
        } else {
          document.getElementsByTagName("head")[0].appendChild(iconLink);
        }
        
        var shortcutLink = document.createElement("link");
        shortcutLink.rel = "shortcut icon";
        shortcutLink.href = absoluteFaviconUrl;
        lockLink(shortcutLink);
        if (document.head) {
          document.head.appendChild(shortcutLink);
        } else {
          document.getElementsByTagName("head")[0].appendChild(shortcutLink);
        }
      } else {
        links.forEach(function(link) {
          lockLink(link);
          
          // One-time inline sync in case it was already altered before we locked it
          var normCurrent = (link.href || "").split("?")[0].replace(/^https?:/, "").toLowerCase();
          var normTarget = (absoluteFaviconUrl || "").split("?")[0].replace(/^https?:/, "").toLowerCase();
          if (normCurrent !== normTarget) {
            try {
              link.href = absoluteFaviconUrl;
            } catch(e) {}
          }
        });
      }
    }

    apply();
    // Run on key loading stages
    document.addEventListener("DOMContentLoaded", apply);
    window.addEventListener("load", apply);
    
    // Set up MutationObserver to block child and attribute attempts to overwrite our favicon!
    var target = document.documentElement;
    if (target && typeof MutationObserver !== 'undefined') {
      var observer = new MutationObserver(function(mutations) {
        var needsApply = false;
        mutations.forEach(function(mutation) {
          if (mutation.type === 'childList') {
            var addedNodes = Array.prototype.slice.call(mutation.addedNodes);
            var hasIcon = addedNodes.some(function(node) {
              return node.nodeType === 1 && node.tagName.toUpperCase() === 'LINK' && node.getAttribute('rel') && node.getAttribute('rel').indexOf('icon') !== -1;
            });
            if (hasIcon) needsApply = true;
          } else if (mutation.type === 'attributes') {
            var node = mutation.target;
            if (node && node.nodeType === 1 && node.tagName.toUpperCase() === 'LINK') {
              var rel = node.getAttribute('rel');
              if (rel && rel.indexOf('icon') !== -1) {
                var normCurrent = (node.href || "").split("?")[0].replace(/^https?:/, "").toLowerCase();
                if (normCurrent !== normTarget) {
                  needsApply = true;
                }
              }
            }
          }
        });
        if (needsApply) {
          apply();
        }
      });
      observer.observe(target, { 
        childList: true, 
        subtree: true, 
        attributes: true
      });
    }
    
    // 200ms high-frequency heartbeat sweep to guarantee absolute victory over FitNesse scripts
    setInterval(apply, 200);
  }

    function iciBuildRow(action, values) {
      var tokens = action.s.slice();
      var optional = {};
      action.p.forEach(function(param) { if (param[3] === "opt") { optional[param[0]] = true; } });
      var cells = [];
      for (var i = 0; i < tokens.length; i++) {
        var match = /^\{(\w+)\}$/.exec(tokens[i]);
        if (match) {
          var value = (values[match[1]] || "").trim();
          if (value === "" && optional[match[1]]) {
            if (cells.length > 1) { cells.pop(); }   // drop the joining keyword (timeout / on / with)
            continue;
          }
          cells.push(value);
        } else {
          cells.push(tokens[i]);
        }
      }
      return "| " + cells.join(" | ") + " |";
    }

      return null;
    }

    function iciSelectedPage() {
      var select = iciEl("ici-builder-page");
      return select ? select.value : "";
    }

    function iciPageLocators() {
      return iciPages[iciSelectedPage()] || {};
    }

    function iciCurrentValues(action) {
      var values = {};
      action.p.forEach(function(param) {
        var select = iciEl("ici-builder-param-" + param[0] + "-sel");
        var input = iciEl("ici-builder-param-" + param[0]);
        if (select && select.style.display !== "none" && select.value !== "__custom__") {
          var selectedPage = iciSelectedPage();
          values[param[0]] = selectedPage === "LoginPage"
            ? select.value
            : selectedPage + "." + select.value;
        } else {
          values[param[0]] = input ? input.value : "";
        }
      });
      return values;
    }

    function iciInitBuilder() {
      var categorySelect = iciEl("ici-builder-category");
      if (!categorySelect || categorySelect.options.length) { return; }
      var seen = {};
      ICI_ACTIONS.forEach(function(action) {
        if (!seen[action.c]) {
          seen[action.c] = true;
          categorySelect.appendChild(new Option(action.c, action.c));
        }
      });
      iciRenderPageOptions();
      iciOnBuilderCategoryChange();
      iciRenderBuilderSteps();
    }

    function iciRenderPageOptions() {
      var select = iciEl("ici-builder-page");
      if (!select) { return; }
      var previous = select.value;
      select.innerHTML = "";
      select.appendChild(new Option("(none - type selectors)", ""));
      Object.keys(iciPages).sort().forEach(function(name) { select.appendChild(new Option(name, name)); });
      if (previous && iciPages[previous]) { select.value = previous; }
      else if (!previous && Object.keys(iciPages).length === 1) { select.value = Object.keys(iciPages)[0]; }
      var list = iciEl("ici-manage-pages");
      if (list) {
        list.innerHTML = "";
        Object.keys(iciPages).sort().forEach(function(name) { list.appendChild(new Option(name, name)); });
      }
      iciRenderManageList();
    }

    window.iciLoadPages = function() {
      return fetch(ICI_PAGES_URL).then(function(response) { return response.json(); }).then(function(data) {
        iciPages = (data && data.pages) || {};
      }).catch(function() {
        iciPages = {};
      }).then(function() {
        iciRenderPageOptions();
        if (iciEl("ici-builder-action") && iciEl("ici-builder-action").options.length) { iciOnBuilderActionChange(); }
      });
    };

    window.iciOnBuilderPageChange = function() {
      iciOnBuilderActionChange();
      var manage = iciEl("ici-manage-page");
      if (manage && iciSelectedPage()) { manage.value = iciSelectedPage(); }
      iciRenderManageList();
    };

    window.iciOnBuilderCategoryChange = function() {
      var category = iciEl("ici-builder-category").value;
      var actionSelect = iciEl("ici-builder-action");
      actionSelect.innerHTML = "";
      ICI_ACTIONS.forEach(function(action) {
        if (action.c === category) { actionSelect.appendChild(new Option(action.l, action.l)); }
      });
      iciOnBuilderActionChange();
    };

    window.iciOnBuilderActionChange = function() {
      var action = iciCurrentAction();
      var holder = iciEl("ici-builder-params");
      var kept = {};
      if (action) { kept = iciCurrentValues(action); }
      holder.innerHTML = "";
      if (!action) { return; }
      var pageLocators = iciPageLocators();
      var locatorNames = Object.keys(pageLocators);
      action.p.forEach(function(param) {
        var field = document.createElement("div");
        field.className = "ici-builder-field";
        var label = document.createElement("label");
        label.className = "ici-builder-label";
        label.innerHTML = param[1] + (param[3] === "opt" ? ' <span class="opt">(optional)</span>' : "");
        field.appendChild(label);

        var input;
        if (param[2] === "select") {
          input = document.createElement("select");
          if (param[3] === "opt") { input.appendChild(new Option("(default)", "")); }
          param[4].forEach(function(choice) { input.appendChild(new Option(choice, choice)); });
        } else {
          input = document.createElement("input");
          input.type = param[2] === "number" ? "number" : "text";
          if (param[2] === "number") { input.min = "0"; input.step = "any"; }
          if (param[2] === "locator") {
            input.placeholder = "selector, e.g. #id, css=..., xpath=..., text=...";
          } else if (typeof param[4] === "string") {
            input.placeholder = param[4];
          }
          if (typeof param[5] === "string") { input.value = (kept[param[0]] !== undefined && kept[param[0]] !== "") ? kept[param[0]] : param[5]; }
        }
        input.id = "ici-builder-param-" + param[0];
        input.className = "ici-env-input";
        input.addEventListener("input", iciGenerateBuilderOutput);
        input.addEventListener("change", iciGenerateBuilderOutput);
        input.addEventListener("keydown", function(event) {
          if (event.key === "Enter") { event.preventDefault(); window.iciAppendStepToScript(); }
        });

        if (param[2] === "locator" && locatorNames.length) {
          var picker = document.createElement("select");
          picker.id = "ici-builder-param-" + param[0] + "-sel";
          picker.className = "ici-env-input";
          locatorNames.forEach(function(name) { picker.appendChild(new Option(name + "   →   " + pageLocators[name], name)); });
          picker.appendChild(new Option("Custom selector...", "__custom__"));
          picker.addEventListener("change", function() {
            input.style.display = picker.value === "__custom__" ? "block" : "none";
            input.style.marginTop = "6px";
            iciGenerateBuilderOutput();
          });
          input.style.display = "none";
          field.appendChild(picker);
        }
        field.appendChild(input);
        holder.appendChild(field);
      });
      iciEl("ici-builder-doc").textContent = action.d || "";
      iciGenerateBuilderOutput();
    };

    window.iciGenerateBuilderOutput = function() {
      var action = iciCurrentAction();
      if (!action) { return; }
      var values = iciCurrentValues(action);
      iciEl("ici-builder-step-preview").textContent = iciBuildRow(action, values);
      var missing = action.p.some(function(param) { return param[3] !== "opt" && !(values[param[0]] || "").trim(); });
      iciEl("ici-builder-add").disabled = missing;
    };

    window.iciAppendStepToScript = function() {
      var action = iciCurrentAction();
      if (!action || iciEl("ici-builder-add").disabled) { return; }
      iciBuilderSteps.push({ row: iciBuildRow(action, iciCurrentValues(action)), kind: action.s[0] });
      action.p.forEach(function(param) {
        var input = iciEl("ici-builder-param-" + param[0]);
        if (input && input.tagName === "INPUT") { input.value = (typeof param[5] === "string") ? param[5] : ""; }
      });
      iciGenerateBuilderOutput();
      iciRenderBuilderSteps();
      iciDrawerToast("Step added");
    };

    function iciRenderBuilderSteps() {
      var list = iciEl("ici-builder-steps");
      if (!list) { return; }
      list.innerHTML = "";
      var depth = 0;
      iciBuilderSteps.forEach(function(item, index) {
        if (item.kind === "end if" || item.kind === "end loop" || item.kind === "else branch") { depth = Math.max(0, depth - 1); }
        var line = document.createElement("div");
        line.className = "ici-builder-step";
        line.innerHTML = '<span class="num"></span><span class="txt"></span>' +
          '<button class="ctl" title="Move up">▲</button><button class="ctl" title="Move down">▼</button><button class="ctl" title="Remove">✕</button>';
        line.querySelector(".num").textContent = (index + 1);
        var text = line.querySelector(".txt");
        text.style.paddingLeft = (depth * 14) + "px";
        text.textContent = item.row;
        var buttons = line.querySelectorAll(".ctl");
        buttons[0].onclick = function() { iciMoveStep(index, -1); };
        buttons[1].onclick = function() { iciMoveStep(index, 1); };
        buttons[2].onclick = function() { iciBuilderSteps.splice(index, 1); iciRenderBuilderSteps(); iciGenerateBuilderOutput(); };
        list.appendChild(line);
        if (item.kind === "if condition" || item.kind === "for each" || item.kind === "else branch") { depth += 1; }
      });
      if (!iciBuilderSteps.length) {
        list.innerHTML = '<div class="ici-builder-empty">No steps yet. Pick an action above and press Add step.</div>';
      }
      iciEl("ici-builder-count").textContent = iciBuilderSteps.length;
      var script = "";
      if (iciBuilderSteps.length) {
        script = "| script | fixtures.UiFixture |\n" + iciBuilderSteps.map(function(item) { return item.row; }).join("\n");
      }
      iciEl("ici-builder-script").value = script;
    }

    function iciMoveStep(index, direction) {
      var target = index + direction;
      if (target < 0 || target >= iciBuilderSteps.length) { return; }
      var moved = iciBuilderSteps.splice(index, 1)[0];
      iciBuilderSteps.splice(target, 0, moved);
      iciRenderBuilderSteps();
      iciGenerateBuilderOutput();
    }

    window.iciClearBuilderScript = function() {
      iciBuilderSteps = [];
      iciRenderBuilderSteps();
      iciGenerateBuilderOutput();
    };

    window.iciCopyBuilderScript = function() {
      var text = iciEl("ici-builder-script").value;
      if (text) {
        iciCopyToClipboard(text);
      } else {
        iciDrawerToast("Add some steps first");
      }
    };

    /* ── Page locator management ── */
    function iciRenderManageList() {
      var holder = iciEl("ici-manage-list");
      var pageInput = iciEl("ici-manage-page");
      if (!holder || !pageInput) { return; }
      var page = pageInput.value.trim();
      var locators = iciPages[page] || {};
      holder.innerHTML = "";
      Object.keys(locators).forEach(function(name) {
        var row = document.createElement("div");
        row.className = "ici-manage-row";
        row.innerHTML = '<span class="n"></span><span class="s"></span><button class="ctl" title="Remove">✕</button>';
        row.querySelector(".n").textContent = name;
        row.querySelector(".s").textContent = locators[name];
        row.querySelector(".ctl").onclick = function() { iciPostPage({ page: page, remove: name }, "Locator removed"); };
        holder.appendChild(row);
      });
    }

    function iciPostPage(payload, doneMessage) {
      return fetch(ICI_PAGES_URL, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload)
      }).then(function(response) { return response.json(); }).then(function(data) {
        if (data.error) { iciDrawerToast("⚠️ " + data.error); return; }
        iciDrawerToast(doneMessage);
        return iciLoadPages();
      }).catch(function() { iciDrawerToast("⚠️ User store server is not reachable"); });
    }

    window.iciSaveLocator = function() {
      var page = iciEl("ici-manage-page").value.trim();
      var name = iciEl("ici-manage-name").value.trim();
      var selector = iciEl("ici-manage-selector").value.trim();
      if (!page || !name || !selector) { iciDrawerToast("Page, locator name and selector are required"); return; }
      iciPostPage({ page: page, locator: name, selector: selector }, "Locator saved").then(function() {
        iciEl("ici-manage-name").value = "";
        iciEl("ici-manage-selector").value = "";
        iciEl("ici-builder-page").value = iciPages[page] ? page : iciEl("ici-builder-page").value;
        iciOnBuilderActionChange();
      });
    };

    window.iciRenderManageList = iciRenderManageList;

    function iciDrawerToast(message) {
      var toast = iciEl("ici-drawer-toast");
      if (!toast) { return; }
      toast.textContent = message;
      toast.classList.add("show");
      clearTimeout(toast._timer);
      toast._timer = setTimeout(function() { toast.classList.remove("show"); }, 1800);
    }

    window.iciCopyToClipboard = function(text) {
      var done = function() { iciDrawerToast("📋 Copied: " + (text.length > 40 ? text.substring(0, 40) + "..." : text).replace(/\n/g, " ")); };
      if (navigator.clipboard) {
        navigator.clipboard.writeText(text).then(done);
      } else {
        var textarea = document.createElement("textarea");
        textarea.value = text;
        document.body.appendChild(textarea);
        textarea.select();
        document.execCommand("copy");
        document.body.removeChild(textarea);
        done();
      }
    };

    window.iciLiveDebug = function() {
      // POST to local backend on 8090 to activate the headless=false override
      fetch("http://127.0.0.1:8090/live-debug", {
        method: "POST"
      })
      .then(function() {
        // Redirect browser to trigger the test run
        location.href = getRunUrl();
      })
      .catch(function(err) {
        console.error("Live debug trigger failed:", err);
        location.href = getRunUrl(); // Fallback to standard run anyway
      });
    };
    
    console.log('%c✓ All 6 UX fixes initialized successfully', 'color: green; font-weight: bold;');
  });

})();
</script>
